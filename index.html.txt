<div class="counter">
    <div class="plus" onclick="addPlus()">🟢 + <span id="plus">0</span></div>
    <div class="minus" onclick="addMinus()">🔴 − <span id="minus">0</span></div>
</div>

<div class="reset" onclick="resetCounter()">↻</div>

<script>
let plus = Number(localStorage.getItem("tradePlus")) || 0;
let minus = Number(localStorage.getItem("tradeMinus")) || 0;

function update() {
    document.getElementById("plus").textContent = plus;
    document.getElementById("minus").textContent = minus;
}

function addPlus() {
    plus++;
    localStorage.setItem("tradePlus", plus);
    update();
}

function addMinus() {
    minus++;
    localStorage.setItem("tradeMinus", minus);
    update();
}

function resetCounter() {
    if (confirm("Скинути лічильник?")) {
        plus = 0;
        minus = 0;
        localStorage.setItem("tradePlus", 0);
        localStorage.setItem("tradeMinus", 0);
        update();
    }
}

update();
</script>
